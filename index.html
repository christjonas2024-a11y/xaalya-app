// Variable globale pour stocker les 294 corridors en mémoire locale let catalogueCorridors = []; // Devise sélectionnée par défaut const DEFAUTS = { source: 'EUR', target: 'XOF', amount: 100 }; /** * Initialisation au chargement de la page */ document.addEventListener('DOMContentLoaded', async () => { await chargerDonneesRates(); if (catalogueCorridors.length > 0) { initialiserSelects(); attacherEcouteursEvenements(); afficherComparaison(); } }); /** * 1. Charge le fichier rates.json généré par le scraper */ async function chargerDonneesRates() { try { const response = await fetch('rates.json?v=' + new Date().getTime()); // Évite le cache navigateur agressif if (!response.ok) throw new Error("Impossible de charger le fichier des taux."); catalogueCorridors = await response.json(); } catch (error) { console.error("Erreur d'initialisation Xaalya :", error); document.getElementById('operatorsList').innerHTML = `⚠️ Impossible de charger les données comparatives en direct. Veuillez réessayer.`; } } /** * 2. Rempli dynamiquement les sélecteurs déroulants (Dropdowns) */ function initialiserSelects() { const selectSource = document.getElementById('sourceCurrency'); const selectTarget = document.getElementById('targetCurrency'); // Extraction unique des devises de départ et d'arrivée const sourcesUniques = [...new Set(catalogueCorridors.map(c => c.devisesDepart))]; const ciblesUniques = [...new Set(catalogueCorridors.map(c => c.deviseArrivee))]; // Remplissage du select Source selectSource.innerHTML = sourcesUniques.map(src => `` ).join(''); // Remplissage du select Cible selectTarget.innerHTML = ciblesUniques.map(tgt => `` ).join(''); } /** * 3. Écouteurs d'événements sur les contrôles utilisateur */ function attacherEcouteursEvenements() { document.getElementById('sourceCurrency').addEventListener('change', afficherComparaison); document.getElementById('targetCurrency').addEventListener('change', afficherComparaison); document.getElementById('sendAmount').addEventListener('input', afficherComparaison); } /** * 4. Recherche du corridor et affichage dynamique des 10 opérateurs */ function afficherComparaison() { const source = document.getElementById('sourceCurrency').value; const target = document.getElementById('targetCurrency').value; const amount = parseFloat(document.getElementById('sendAmount').value) || 0; const tbody = document.getElementById('operatorsList'); const infoBlock = document.getElementById('lastUpdatedInfo'); // Recherche du corridor correspondant aux paires choisies const corridorActif = catalogueCorridors.find(c => c.devisesDepart === source && c.deviseArrivee === target); if (!corridorActif || !corridorActif.offres || corridorActif.offres.length === 0) { tbody.innerHTML = `Aucun résultat disponible pour le couloir ${source} ➔ ${target}.`; infoBlock.textContent = ''; return; } // Affichage de l'horodatage de la dernière mise à jour const dateMaj = new Date(corridorActif.misAJour).toLocaleString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' }); infoBlock.innerHTML = `Taux actualisés le : ${dateMaj}`; // Traitement et calcul pour chaque opérateur const offresCalculees = corridorActif.offres.map(offre => { // Calcul du montant réel reçu par le destinataire const montantRecu = (amount * offre.taux).toFixed(2); return { ...offre, montantRecu: parseFloat(montantRecu) }; }); // Tri de l'offre la plus avantageuse à la moins avantageuse (basé sur le montant reçu) offresCalculees.sort((a, b) => b.montantRecu - a.montantRecu); // Construction du tableau HTML tbody.innerHTML = offresCalculees.map((offre, idx) => { const estLeMeilleur = idx === 0; return ` ${idx + 1} ${offre.operateur} ${estLeMeilleur ? 'Le Moins Cher' : ''} 1 ${source} = ${offre.taux} ${target} ${offre.frais > 0 ? `${offre.frais} ${source}` : 'Gratuit'} ${offre.montantRecu.toLocaleString('fr-FR')} ${target} ${offre.delai || 'Instantané'} Envoyer ➔ `; }).join(''); }
